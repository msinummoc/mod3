# yearly event
country_event = {
	id = mod3.2
	title = mod3.2.t
	desc = mod3.2.d
	picture = INVENTION_eventPicture
	hidden = yes
	is_triggered_only = yes
	trigger = {
		owns = 1
	}
	immediate = {
		# 全局变量
		random_list = {
			9 = {}
			1 = {
				random_list = {
					1 = { set_variable = { which = mod3_age_dev_cost value = -0.3 } }
					6 = { set_variable = { which = mod3_age_dev_cost value = -0.2 } }
					15 = { set_variable = { which = mod3_age_dev_cost value = -0.1 } }
					20 = { set_variable = { which = mod3_age_dev_cost value = 0 } }
					15 = { set_variable = { which = mod3_age_dev_cost value = 0.1 } }
					6 = { set_variable = { which = mod3_age_dev_cost value = 0.2 } }
					1 = { set_variable = { which = mod3_age_dev_cost value = 0.3 } }
				}
				
				if = {
					limit = { current_age = age_of_discovery }
					change_variable = { which = mod3_age_dev_cost value = 0 }
				}
				else_if = {
					limit = { current_age = age_of_reformation }
					change_variable = { which = mod3_age_dev_cost value = -0.05 }
				}
				else_if = {
					limit = { current_age = age_of_absolutism }
					change_variable = { which = mod3_age_dev_cost value = 0.1 }
				}
				else_if = {
					limit = { current_age = age_of_revolutions }
					change_variable = { which = mod3_age_dev_cost value = -0.1 }
				}
			}
			
		}
		
		mod3_dip_dev_yearly_before_effect = yes
		
		every_country = {
			set_variable = { which = mod3_age_dev_cost which = prev }
			country_event = {
				id = mod3.3
			}
		}
		mod3_mil_dev_yearly_2_effect = yes
		every_country = {
			country_event = {
				id = mod3.5
				days = 3
			}
		}
    }
	
	option = {
		
	}
}
