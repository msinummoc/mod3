
can_improve_tax = {
	condition = {
		tooltip = "AI should slow down"
		potential = {
			owner = {
				ai = yes
			}
		}
		allow = {
			or = {
				not = { check_variable = { which = mod3_dev_vs_dev_lag value = 1.2 } }
				and = {
					not = { check_variable = { which = mod3_local_dev_cost value = 1.5 } }
					not = { check_variable = { which = mod3_dev_vs_dev_lag value = 1.5 } }
				}
				not = { development = 15 }
			}
		}
	}
}
