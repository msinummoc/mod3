
can_improve_tax = {
	condition = {
		tooltip = "AI should slow down"
		potential = {
			owner = {
				ai = yes
			}
		}
		allow = {
			or = {
				not = { check_variable = { which = mod3_local_dev_cost value = 1.5 } }

				}
			}
		}
	}

		tooltip = "AI should stop"
		potential = {
			owner = {
				ai = yes
			}
		}
		allow = {
			always = yes
		}
	}
}
