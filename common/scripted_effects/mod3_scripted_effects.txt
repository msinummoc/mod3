mod3_init_effect = {
    if = {
        limit = { not = { is_year = 1445 } }
        if = {
            limit = { tag = KOR }
            capital_scope = {
                add_base_tax = -2
                add_base_production = -2
                add_base_manpower = -1
            }
        }
        else_if = {
            limit = { OR = { tag = CRI tag = GOL tag = KAZ tag = NOG tag = SHY } }
            every_owned_province = {
                if = { limit = { base_tax = 2 } add_base_tax = -1 }
                if = { limit = { base_production = 2 } add_base_production = -1 }
                if = { limit = { base_manpower = 3 } add_base_manpower = -1 }
            }
        }
        else_if = {
            limit = { capital_scope = { superregion = east_indies_superregion } }
            every_owned_province = {
                if = { limit = { base_tax = 3 } add_base_tax = -1 }
                if = { limit = { base_production = 3 } add_base_production = -1 }
                if = { limit = { base_manpower = 3 } add_base_manpower = -1 }
            }
        }
        else_if = {
            limit = { 
                capital_scope = { superregion = india_superregion } 
                development = 50
            }
            capital_scope = {
                add_base_manpower = 1
                add_base_tax = 1
                add_base_production = 1
            }
        }
        else_if = {
            limit = { 
                capital_scope = { superregion = india_superregion } 
                total_development = 50
            }
            capital_scope = {
                add_base_manpower = 1
                add_base_tax = 1
                add_base_production = 1
            }
        }
        else_if = {
            limit = {
                any_owned_province = { culture = polish }
            }
            every_owned_province = {
                limit = { development = 10 culture = polish }
                add_base_production = -1
                add_base_tax = -1
            }
        }
    }
    mod3_yearly_effect = yes
    every_owned_province = {
        change_variable = { which = mod3_dev_momentum value = 0.3 }
        if = {
            limit = { not = { check_variable = { which = mod3_dev_lag value = 1 } } }
            set_variable = { which = mod3_dev_lag which = mod3_dev }
            change_variable = { which = mod3_dev_lag value = 0.1 }
        }
    }
}

mod3_yearly_effect = {
    if = {
        limit = { not = { has_country_modifier = mod3_less_governing_capacity } }
        add_country_modifier = {
            name = mod3_less_governing_capacity
            duration = -1
        }
    }
    set_variable = { which = mod3_num_perm_claim value = 0 }
    every_country = {
        limit = {
            prev = {
                or = {
                    claim = prev
                    core_claim = prev
                }
            }
        }
        if = {
            limit = {
                prev = { core_claim = prev }
            }
            prev = { change_variable = { which = mod3_num_perm_claim value = 1 } }
        }
        else_if = {
            limit = {
                any_owned_province = {
                    is_permanent_claim = prev
                }
            }
            prev = { change_variable = { which = mod3_num_perm_claim value = 1 } }
        }
    }
    
    every_owned_province = {
        # 每个省份的贡献系数
        export_to_variable = { which = mod3_dev_buff_modifier value = trigger_value:development }
        divide_variable = { which = mod3_dev_buff_modifier value = 10 } 
        round_variable = { which = mod3_dev_buff_modifier value = -1 } # 向下取整
        if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 7 } }
            multiply_variable = { which = mod3_dev_buff_modifier value = 0.2 }
            change_variable = { which = mod3_dev_buff_modifier value = 1.1 }
        }
        else_if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 6 } }
            set_variable = { which = mod3_dev_buff_modifier value = 2.25 }
        }
        else_if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 5 } }
            set_variable = { which = mod3_dev_buff_modifier value = 2.0 }
        }
        else_if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 4 } }
            set_variable = { which = mod3_dev_buff_modifier value = 1.75 }
        }
        else_if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 3 } }
            set_variable = { which = mod3_dev_buff_modifier value = 1.5 }
        }
        else_if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 2 } }
            set_variable = { which = mod3_dev_buff_modifier value = 1.1 }
        }
        else_if = {
            limit = { check_variable = { which = mod3_dev_buff_modifier value = 1 } }
            set_variable = { which = mod3_dev_buff_modifier value = 0.7 }
        }
        else = {
            set_variable = { which = mod3_dev_buff_modifier value = 0.3 }
        }

        # 荒废度降低贡献值 100%荒废度导致不贡献 对人力贡献不产生影响
        if = {
            limit = { devastation = 1 }
            export_to_variable = { which = mod3_devast value = trigger_value:devastation }
            subtract_variable = { which = mod3_devast value = 100 }
            multiply_variable = { which = mod3_devast value = -0.01 }
            multiply_variable = { which = mod3_dev_buff_modifier which = mod3_devast }
        }
    }

    mod3_adm_dev_yearly_effect = yes
    mod3_dip_dev_yearly_1_effect = yes
    mod3_mil_dev_yearly_1_effect = yes
    country_event = {
        id = mod3.4
        days = 3
    }
}

mod3_adm_dev_yearly_effect = {
    every_owned_province = {
        clr_province_flag = mod3_adm_1_flag
    }
    every_owned_province = {
        limit = { not = { has_province_flag = mod3_adm_1_flag } }
        export_to_variable = { which = mod3_adm_dev value = trigger_value:base_tax }
        multiply_variable = { which = mod3_adm_dev which = mod3_dev_buff_modifier }
        area = {
            limit = { and = { not = { province_id = prev } owned_by = root } }
            export_to_variable = { which = mod3_adm_dev value = trigger_value:base_tax }
            multiply_variable = { which = mod3_adm_dev which = mod3_dev_buff_modifier }
            prev = { change_variable = { which = mod3_adm_dev which = prev } }
            set_province_flag = mod3_adm_2_flag
            change_variable = { which = mod3_adm_province_number value = 1 }
        }
        set_province_flag = mod3_adm_2_flag
        change_variable = { which = mod3_adm_province_number value = 1 }
        area = {
            limit = { owned_by = root }
            set_variable = { which = mod3_adm_dev which = prev }
            change_variable = { which = mod3_adm_dev_buff which = mod3_adm_dev }
            every_neighbor_province = {
                if = {
                    limit = { and = { owned_by = root not = { has_province_flag = mod3_adm_2_flag } } }
                    set_variable = { which = mod3_adm_dev which = prev }
                    change_variable = { which = mod3_adm_dev_buff which = mod3_adm_dev }
                    set_province_flag = mod3_adm_2_flag
                    change_variable = { which = mod3_adm_province_number value = 1 }
                }
                else_if = {
                    limit = { not = { has_province_flag = mod3_adm_2_flag } }
                    set_variable = { which = mod3_adm_dev which = prev }
                    set_variable = { which = var1 value = 0.5 }
                    multiply_variable = { which = var1 which = mod3_adm_dev }
                    change_variable = { which = mod3_adm_dev_buff which = var1 }
                    set_province_flag = mod3_adm_2_flag
                    change_variable = { which = mod3_adm_province_number value = 0.5 }
                }
            }
            if = {
                limit = { has_port = yes }
                sea_zone = {
                    save_event_target_as = mod3_sea_zone
                    if = {
                        limit = { not = { has_province_flag = mod3_adm_2_flag } }
                        set_variable = { which = mod3_adm_dev which = prev }
                        every_neighbor_province = {
                            if = {
                                limit = { 
                                    and = {
                                        owned_by = root 
                                        not = { has_province_flag = mod3_adm_2_flag } 
                                        sea_zone = { province_id = event_target:mod3_sea_zone }
                                    }
                                }
                                set_variable = { which = mod3_adm_dev which = prev }
                                # 海上传播系数低一点
                                set_variable = { which = var1 value = 0.5 }
                                multiply_variable = { which = var1 which = mod3_adm_dev }
                                change_variable = { which = mod3_adm_dev_buff which = var1 }
                                set_province_flag = mod3_adm_2_flag
                                change_variable = { which = mod3_adm_province_number value = 0.5 }
                            }
                            else_if = {
                                limit = { 
                                    and = { 
                                        not = { has_province_flag = mod3_adm_2_flag } 
                                        sea_zone = { province_id = event_target:mod3_sea_zone }
                                    }    
                                }
                                set_variable = { which = mod3_adm_dev which = prev }
                                set_variable = { which = var1 value = 0.25 }
                                multiply_variable = { which = var1 which = mod3_adm_dev }
                                change_variable = { which = mod3_adm_dev_buff which = var1 }
                                set_province_flag = mod3_adm_2_flag
                                change_variable = { which = mod3_adm_province_number value = 0.25 }
                            }
                        }
                        set_province_flag = mod3_adm_2_flag
                    }
                }
            }
        }
        area = {
            limit = { owned_by = root }
            if = {
                limit = { has_port = yes }
                sea_zone = {
                    every_neighbor_province = {
                        limit = { has_province_flag = mod3_adm_2_flag }
                        clr_province_flag = mod3_adm_2_flag
                    }
                    clr_province_flag = mod3_adm_2_flag
                }
            }
            every_neighbor_province = {
                limit = { has_province_flag = mod3_adm_2_flag }
                clr_province_flag = mod3_adm_2_flag
            }
            clr_province_flag = mod3_adm_2_flag
            set_province_flag = mod3_adm_1_flag
        }
    }
}

mod3_dip_dev_yearly_before_effect = {
    while = {
        limit = {
            any_trade_node = {
                not = { check_variable = { which = mod3_trade_node_loop value = 1 } }
            }
        }
        random_trade_node = {
            limit = {
                not = { check_variable = { which = mod3_trade_node_loop value = 1 } }
            }
            set_variable = { which = mod3_trade_node_dip_dev_buff_0 value = 0 }
            set_variable = { which = mod3_trade_node_province_trade_power value = 0 }
            every_trade_node_member_province = {
                limit = { is_city = yes }
                export_to_variable = { which = mod3_dip_dev_buff_0 value = trigger_value:base_production }
                multiply_variable = { which = mod3_dip_dev_buff_0 which = mod3_dev_buff_modifier }
                export_to_variable = { which = mod3_province_trade_power value = trigger_value:province_trade_power }
                if = {
                    limit = { is_owned_by_trade_company = yes }
                    multiply_variable = { which = mod3_province_trade_power value = 0.25 }
                }
                export_to_variable = { which = mod3_dev value = trigger_value:development }
                set_variable = { which = var2 which = mod3_dev }
                multiply_variable = { which = var2 which = mod3_dev }
                multiply_variable = { which = var2 which = mod3_dev }
                divide_variable = { which = var2 value = 200 }

                change_variable = { which = mod3_province_trade_power which = var2 }

                if = {
                    limit = { province_id = prev }
                    change_variable = { which = mod3_trade_node_dip_dev_buff_0 which = mod3_dip_dev_buff_0 }
                    change_variable = { which = mod3_trade_node_province_trade_power which = mod3_province_trade_power }
                }
                else = {
                    set_variable = { which = mod3_trade_node_dip_dev_buff_0 which = mod3_dip_dev_buff_0 }
                    set_variable = { which = mod3_trade_node_province_trade_power which = mod3_province_trade_power }
                    prev = {
                        change_variable = { which = mod3_trade_node_dip_dev_buff_0 which = prev }
                        change_variable = { which = mod3_trade_node_province_trade_power which = prev }
                    }
                    set_variable = { which = mod3_trade_node_dip_dev_buff_0 value = 0 }
                    set_variable = { which = mod3_trade_node_province_trade_power value = 0 }
                }
            }
            set_variable = { which = mod3_trade_node_loop value = 1 }
        }
    }
    
    while = {
        limit = {
            any_trade_node = {
                check_variable = { which = mod3_trade_node_loop value = 1 } 
            }
        }
        random_trade_node = {
            limit = {
                check_variable = { which = mod3_trade_node_loop value = 1 } 
            }
            set_variable = { which = mod3_trade_node_loop value = 0 }
        }
    }
}

# 按国家结算 对于每个国家，考虑所有拥有贸易力量的节点中的所有省份，并且乘以贸易节点力量占比作为权重 该国自己拥有的省份获得*1.25的权重
mod3_dip_dev_yearly_1_effect = {
    save_event_target_as = mod3_dip_dev_country
    set_variable = { which = mod3_dip_country_numerator value = 0 }
    set_variable = { which = mod3_dip_country_denominator value = 0 }


    while = {
        limit = {
            any_trade_node = {
                and = {
                    trade_share = {
                        country = event_target:mod3_dip_dev_country
                        share = 0.1
                    }
                    not = { check_variable = { which = mod3_trade_node_loop value = 1 } }
                }
            }
        }
        random_trade_node = {
            limit = {
                and = {
                    trade_share = {
                        country = event_target:mod3_dip_dev_country
                        share = 0.1
                    }
                    not = { check_variable = { which = mod3_trade_node_loop value = 1 } }
                }
            }

            mod3_trade_share_calc_effect = yes
            if = {
                limit = { not = { check_variable = { which = mod3_trade_node_trade_share value = 100 } } }
                change_variable = { which = mod3_trade_node_trade_share value = 0.05 }
            }
            set_variable = { which = mod3_trade_node_province_trade_power_025 value = 0 }
            every_trade_node_member_province = {
                limit = { and = { is_city = yes owned_by = event_target:mod3_dip_dev_country } }
                set_variable = { which = mod3_province_trade_power_025 value = 0.25 }
                multiply_variable = { which = mod3_province_trade_power_025 which = mod3_province_trade_power }
                if = {
                    limit = { province_id = prev }
                    change_variable = { which = mod3_trade_node_province_trade_power_025 which = mod3_province_trade_power_025 }
                }
                else = {
                    set_variable = { which = mod3_trade_node_province_trade_power_025 which = mod3_province_trade_power_025 }
                    prev = {
                        change_variable = { which = mod3_trade_node_province_trade_power_025 which = prev }
                    }
                    set_variable = { which = mod3_trade_node_province_trade_power_025 value = 0 }
                }
            }
            set_variable = { which = mod3_trade_node_numerator which = mod3_trade_node_dip_dev_buff_0 }
            divide_variable = { which = mod3_trade_node_numerator value = 100 } # 避免乘以贸易份额后数值溢出
            multiply_variable = { which = mod3_trade_node_numerator which = mod3_trade_node_trade_share }
            set_variable = { which = mod3_trade_node_denominator which = mod3_trade_node_province_trade_power_025 }
            change_variable = { which = mod3_trade_node_denominator which = mod3_trade_node_province_trade_power }
            divide_variable = { which = mod3_trade_node_denominator value = 100 }
            multiply_variable = { which = mod3_trade_node_denominator which = mod3_trade_node_trade_share }
            set_variable = { which = mod3_dip_country_numerator which = mod3_trade_node_numerator }
            set_variable = { which = mod3_dip_country_denominator which = mod3_trade_node_denominator }
            prev = {
                change_variable = { which = mod3_dip_country_numerator which = prev }
                change_variable = { which = mod3_dip_country_denominator which = prev }
            }
            set_variable = { which = mod3_dip_country_numerator value = 0 }
            set_variable = { which = mod3_dip_country_denominator value = 0 }
            set_variable = { which = mod3_trade_node_loop value = 1 }
        }
    }
    set_variable = { which = mod3_dip_dev_buff_1 value = 5 }
    multiply_variable = { which = mod3_dip_dev_buff_1 which = mod3_dip_country_numerator }
    divide_variable = { which = mod3_dip_dev_buff_1 which = mod3_dip_country_denominator }

    
    while = {
        limit = {
            any_trade_node = {
                check_variable = { which = mod3_trade_node_loop value = 1 } 
            }
        }
        random_trade_node = {
            limit = {
                check_variable = { which = mod3_trade_node_loop value = 1 } 
            }
            set_variable = { which = mod3_province_trade_share which = mod3_trade_node_trade_share }
            every_trade_node_member_province = {
                limit = { is_city = yes }
                set_variable = { which = mod3_dip_dev_buff_1 which = event_target:mod3_dip_dev_country }
                multiply_variable = { which = mod3_dip_dev_buff_1 which = mod3_province_trade_power }
                set_variable = { which = mod3_province_trade_share which = prev }
                multiply_variable = { which = mod3_dip_dev_buff_1 which = mod3_province_trade_share }
                divide_variable = { which = mod3_dip_dev_buff_1 value = 100 } 
                if = {
                    limit = { owned_by = event_target:mod3_dip_dev_country }
                    multiply_variable = { which = mod3_dip_dev_buff_1 value = 1.25 }
                }
                change_variable = { which = mod3_dip_dev_buff which = mod3_dip_dev_buff_1 }
            }
            set_variable = { which = mod3_trade_node_loop value = 0 }
        }
    }
}

mod3_mil_dev_yearly_2_effect = {
    every_province = {
        clr_province_flag = mod3_mil_flag
    }
    every_province = {
        limit = { and = {is_city = yes not = { has_province_flag = mod3_mil_flag } } }
        export_to_variable = { which = mod3_mil_dev_buff_2 value = trigger_value:base_manpower }
        set_variable = { which = mod3_region_province_number value = 1 }
        region = {
            limit = { not = { province_id = prev } is_city = yes }
            export_to_variable = { which = mod3_mil_dev_buff_2 value = trigger_value:base_manpower }
            # 这里不乘系数，否则后期同属一个地区的省份发展度会暴涨
            # multiply_variable = { which = mod3_mil_dev_buff_2 which = mod3_dev_buff_modifier }
            prev = {
                change_variable = { which = mod3_mil_dev_buff_2 which = prev }
                change_variable = { which = mod3_region_province_number value = 1 }
            }
        }

        # 除以省份数量的平方
        sqrt_variable = { which = mod3_region_province_number }
        divide_variable = { which = mod3_mil_dev_buff_2 which = mod3_region_province_number }
        multiply_variable = { which = mod3_mil_dev_buff_2 value = 0.3 }

        region = {
            set_variable = { which = mod3_mil_dev_buff_2 which = prev }
            set_province_flag = mod3_mil_flag

# 用7个50%的事件模拟精度到1/128的概率。输入是mod3_dice_helper_input（0-128），输出是mod3_dice_helper_output（1为发生，0为不发生）
mod3_dice_helper_effect = {
    set_variable = { which = rng128_threshold which = mod3_dice_helper_input }

    # 四舍五入到最近整数
    round_variable = { which = rng128_threshold value = 0 }

    # 生成均匀随机整数 1~128
    set_variable = { which = rng128_roll value = 1 }

    random = { chance = 50 change_variable = { which = rng128_roll value = 1 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 2 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 4 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 8 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 16 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 32 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 64 } }

    # 默认失败
    set_variable = { which = mod3_dice_helper_output value = 0 }

    # threshold >= roll 则成功
    if = {
        limit = { check_variable = { which = rng128_threshold which = rng128_roll } }
        set_variable = { which = mod3_dice_helper_output value = 1 }
    }
}
