mod3_init_effect = {
    if = {
        limit = { tag = KOR }
        capital_scope = {
            add_base_tax = -2
            add_base_production = -2
            add_base_manpower = -1
        }
    }
    else_if = {
        limit = { OR = { tag = CRI tag = GOL tag = KAZ tag = NOG tag = SHY } }
        every_owned_province = {
            if = { limit = { base_tax = 2 } add_base_tax = -1 }
            if = { limit = { base_production = 2 } add_base_production = -1 }
            if = { limit = { base_manpower = 3 } add_base_manpower = -1 }
        }
    }
    else_if = {
        limit = { capital_scope = { superregion = east_indies_superregion } }
        every_owned_province = {
            if = { limit = { base_tax = 3 } add_base_tax = -1 }
            if = { limit = { base_production = 3 } add_base_production = -1 }
            if = { limit = { base_manpower = 3 } add_base_manpower = -1 }
        }
    }
    mod3_yearly_effect = yes
}

mod3_adm_dev_yearly_effect = {
    every_owned_province = {
        clr_province_flag = mod3_adm_1_flag
    }
    every_owned_province = {
        limit = { not = { has_province_flag = mod3_adm_1_flag } }
        export_to_variable = { which = mod3_adm_dev value = trigger_value:base_tax }
        area = {
            limit = { and = { not = { province_id = prev } owned_by = root } }
            export_to_variable = { which = mod3_adm_dev value = trigger_value:base_tax }
            prev = { change_variable = { which = mod3_adm_dev which = prev } }
            set_province_flag = mod3_adm_2_flag
        }
        area = {
            limit = { owned_by = root }
            set_variable = { which = mod3_adm_dev which = prev }
            change_variable = { which = mod3_adm_dev_buff which = mod3_adm_dev }
            every_neighbor_province = {
                if = {
                    limit = { and = { owned_by = root not = { has_province_flag = mod3_adm_2_flag } } }
                    set_variable = { which = mod3_adm_dev which = prev }
                    change_variable = { which = mod3_adm_dev_buff which = mod3_adm_dev }
                    set_province_flag = mod3_adm_2_flag
                }
                else_if = {
                    limit = { not = { has_province_flag = mod3_adm_2_flag } }
                    set_variable = { which = mod3_adm_dev which = prev }
                    set_variable = { which = var1 value = 0.5 }
                    multiply_variable = { which = var1 which = mod3_adm_dev }
                    change_variable = { which = mod3_adm_dev_buff which = var1 }
                    set_province_flag = mod3_adm_2_flag
                }
            }
            if = {
                limit = { has_port = yes }
                sea_zone = {
                    if = {
                        limit = { not = { has_province_flag = mod3_adm_2_flag } }
                        set_variable = { which = mod3_adm_dev which = prev }
                        every_neighbor_province = {
                            if = {
                                limit = { 
                                    and = {
                                        owned_by = root 
                                        not = { has_province_flag = mod3_adm_2_flag } 
                                        sea_zone = { province_id = prev }
                                    }
                                }
                                set_variable = { which = mod3_adm_dev which = prev }
                                # 海上传播系数低一点
                                set_variable = { which = var1 value = 0.5 }
                                multiply_variable = { which = var1 which = mod3_adm_dev }
                                change_variable = { which = mod3_adm_dev_buff which = var1 }
                                set_province_flag = mod3_adm_2_flag
                            }
                            else_if = {
                                limit = { 
                                    and = { 
                                        not = { has_province_flag = mod3_adm_2_flag } 
                                        sea_zone = { province_id = prev }
                                    }    
                                }
                                set_variable = { which = mod3_adm_dev which = prev }
                                set_variable = { which = var1 value = 0.25 }
                                multiply_variable = { which = var1 which = mod3_adm_dev }
                                change_variable = { which = mod3_adm_dev_buff which = var1 }
                                set_province_flag = mod3_adm_2_flag
                            }
                        }
                        set_province_flag = mod3_adm_2_flag
                    }
                }
            }
        }
        area = {
            limit = { owned_by = root }
            if = {
                limit = { has_port = yes }
                sea_zone = {
                    every_neighbor_province = {
                        limit = { has_province_flag = mod3_adm_2_flag }
                        clr_province_flag = mod3_adm_2_flag
                    }
                    clr_province_flag = mod3_adm_2_flag
                }
            }
            every_neighbor_province = {
                limit = { has_province_flag = mod3_adm_2_flag }
                clr_province_flag = mod3_adm_2_flag
            }
            clr_province_flag = mod3_adm_2_flag
            set_province_flag = mod3_adm_1_flag
        }
    }
}

mod3_mil_dev_effect = {
    every_province = {
        clr_province_flag = mod3_mil_flag
    }
    every_province = {
        limit = { not = { has_province_flag = mod3_mil_flag }}
        export_to_variable = { which = mod3_mil_dev_buff value = trigger_value:base_manpower }
        set_variable = { which = mod3_region_province_number value = 1 }
        region = {
            limit = { not = { province_id = prev } is_city = yes }
            export_to_variable = { which = mod3_mil_dev_buff value = trigger_value:base_manpower }
            prev = {
                change_variable = { which = mod3_mil_dev_buff which = prev }
                change_variable = { which = mod3_region_province_number value = 1 }
            }
        }
        # 使用 13/(x+40) 估计 1/sprt(x)
        set_variable = { which = var1 value = 13 }
        change_variable = { which = mod3_region_province_number value = 40 }
        divide_variable = { which = var1 which = mod3_region_province_number }
        multiply_variable = { which = mod3_mil_dev_buff which = var1 }
        multiply_variable = { which = mod3_mil_dev_buff value = 1.5 }
        region = {
            set_variable = { which = mod3_mil_dev_buff which = prev }
            set_province_flag = mod3_mil_flag
        }
    }
}
