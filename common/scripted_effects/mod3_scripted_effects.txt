mod3_init_effect = {
    if = {
        limit = { not = { is_year = 1445 } }
        if = {
            limit = { tag = KOR }
            capital_scope = {
                add_base_tax = -2
                add_base_production = -2
                add_base_manpower = -1
            }
        }
        else_if = {
            limit = { OR = { tag = CRI tag = GOL tag = KAZ tag = NOG tag = SHY } }
            every_owned_province = {
                if = { limit = { base_tax = 2 } add_base_tax = -1 }
                if = { limit = { base_production = 2 } add_base_production = -1 }
                if = { limit = { base_manpower = 3 } add_base_manpower = -1 }
            }
        }
        else_if = {
            limit = { capital_scope = { superregion = east_indies_superregion } }
            every_owned_province = {
                if = { limit = { base_tax = 3 } add_base_tax = -1 }
                if = { limit = { base_production = 3 } add_base_production = -1 }
                if = { limit = { base_manpower = 3 } add_base_manpower = -1 }
            }
        }
    }
    mod3_yearly_effect = yes
    every_owned_province = {
        change_variable = { which = mod3_dev_momentum value = 0.3 }
    }
}

mod3_adm_dev_yearly_effect = {
    every_owned_province = {
        clr_province_flag = mod3_adm_1_flag
    }
    every_owned_province = {
        limit = { not = { has_province_flag = mod3_adm_1_flag } }
        export_to_variable = { which = mod3_adm_dev value = trigger_value:base_tax }
        multiply_variable = { which = mod3_adm_dev which = mod3_dev_buff_modifier }
        area = {
            limit = { and = { not = { province_id = prev } owned_by = root } }
            export_to_variable = { which = mod3_adm_dev value = trigger_value:base_tax }
            multiply_variable = { which = mod3_adm_dev which = mod3_dev_buff_modifier }
            prev = { change_variable = { which = mod3_adm_dev which = prev } }
            set_province_flag = mod3_adm_2_flag
            change_variable = { which = mod3_adm_province_number value = 1 }
        }
        set_province_flag = mod3_adm_2_flag
        change_variable = { which = mod3_adm_province_number value = 1 }
        area = {
            limit = { owned_by = root }
            set_variable = { which = mod3_adm_dev which = prev }
            change_variable = { which = mod3_adm_dev_buff which = mod3_adm_dev }
            every_neighbor_province = {
                if = {
                    limit = { and = { owned_by = root not = { has_province_flag = mod3_adm_2_flag } } }
                    set_variable = { which = mod3_adm_dev which = prev }
                    change_variable = { which = mod3_adm_dev_buff which = mod3_adm_dev }
                    set_province_flag = mod3_adm_2_flag
                    change_variable = { which = mod3_adm_province_number value = 1 }
                }
                else_if = {
                    limit = { not = { has_province_flag = mod3_adm_2_flag } }
                    set_variable = { which = mod3_adm_dev which = prev }
                    set_variable = { which = var1 value = 0.5 }
                    multiply_variable = { which = var1 which = mod3_adm_dev }
                    change_variable = { which = mod3_adm_dev_buff which = var1 }
                    set_province_flag = mod3_adm_2_flag
                    change_variable = { which = mod3_adm_province_number value = 0.5 }
                }
            }
            if = {
                limit = { has_port = yes }
                sea_zone = {
                    save_event_target_as = mod3_sea_zone
                    if = {
                        limit = { not = { has_province_flag = mod3_adm_2_flag } }
                        set_variable = { which = mod3_adm_dev which = prev }
                        every_neighbor_province = {
                            if = {
                                limit = { 
                                    and = {
                                        owned_by = root 
                                        not = { has_province_flag = mod3_adm_2_flag } 
                                        sea_zone = { province_id = event_target:mod3_sea_zone }
                                    }
                                }
                                set_variable = { which = mod3_adm_dev which = prev }
                                # 海上传播系数低一点
                                set_variable = { which = var1 value = 0.5 }
                                multiply_variable = { which = var1 which = mod3_adm_dev }
                                change_variable = { which = mod3_adm_dev_buff which = var1 }
                                set_province_flag = mod3_adm_2_flag
                                change_variable = { which = mod3_adm_province_number value = 0.5 }
                            }
                            else_if = {
                                limit = { 
                                    and = { 
                                        not = { has_province_flag = mod3_adm_2_flag } 
                                        sea_zone = { province_id = event_target:mod3_sea_zone }
                                    }    
                                }
                                set_variable = { which = mod3_adm_dev which = prev }
                                set_variable = { which = var1 value = 0.25 }
                                multiply_variable = { which = var1 which = mod3_adm_dev }
                                change_variable = { which = mod3_adm_dev_buff which = var1 }
                                set_province_flag = mod3_adm_2_flag
                                change_variable = { which = mod3_adm_province_number value = 0.25 }
                            }
                        }
                        set_province_flag = mod3_adm_2_flag
                    }
                }
            }
        }
        area = {
            limit = { owned_by = root }
            if = {
                limit = { has_port = yes }
                sea_zone = {
                    every_neighbor_province = {
                        limit = { has_province_flag = mod3_adm_2_flag }
                        clr_province_flag = mod3_adm_2_flag
                    }
                    clr_province_flag = mod3_adm_2_flag
                }
            }
            every_neighbor_province = {
                limit = { has_province_flag = mod3_adm_2_flag }
                clr_province_flag = mod3_adm_2_flag
            }
            clr_province_flag = mod3_adm_2_flag
            set_province_flag = mod3_adm_1_flag
        }
    }
}

# 用7个50%的事件模拟精度到1/128的概率。输入是mod3_dice_helper_input（0-128），输出是mod3_dice_helper_output（1为发生，0为不发生）
mod3_dice_helper_effect = {
    set_variable = { which = rng128_threshold which = mod3_dice_helper_input }

    # 四舍五入到最近整数
    round_variable = { which = rng128_threshold value = 0 }

    # 生成均匀随机整数 1~128
    set_variable = { which = rng128_roll value = 1 }

    random = { chance = 50 change_variable = { which = rng128_roll value = 1 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 2 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 4 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 8 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 16 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 32 } }
    random = { chance = 50 change_variable = { which = rng128_roll value = 64 } }

    # 默认失败
    set_variable = { which = mod3_dice_helper_output value = 0 }

    # threshold >= roll 则成功
    if = {
        limit = { check_variable = { which = rng128_threshold which = rng128_roll } }
        set_variable = { which = mod3_dice_helper_output value = 1 }
    }
}
